{{/* Visible when/where for an event page. Recurring events show `schedule` instead of the front-matter date (that date is a start marker, not the next occurrence). One-off events show the date. Expects a page as context. */}} {{- $dateFormat := site.Params.events.dateFormat | default "2 January, 2006" -}} {{- $hasSchedule := .Params.schedule -}} {{- $hasDate := and (not .Params.recurring) (not .Date.IsZero) -}} {{- $location := .Params.location_name -}} {{- with .Params.location_locality -}} {{- if $location -}} {{- $location = printf "%s, %s" $location . -}} {{- else -}} {{- $location = . -}} {{- end -}} {{- end -}} {{- if or $hasSchedule $hasDate $location -}}
{{- if $hasSchedule -}}
{{ lang.Translate "eventWhen" }}:
{{ .Params.schedule }}
{{- else if $hasDate -}}
{{ lang.Translate "eventDate" }}:
{{ time.Format $dateFormat .Date }}
{{- end -}} {{- with $location -}}
{{ lang.Translate "eventWhere" }}:
{{ . }}
{{- end -}}
{{- end -}}
